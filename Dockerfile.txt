# Stufe 1: Wir holen uns das fertige Terminal-Binary aus dem ttyd-Image
FROM tsl0922/ttyd:latest AS ttyd

# Stufe 2: Wir nutzen das stabile, super-schlanke Java 21 Image
FROM eclipse-temurin:21-jre-alpine

# Kopiere das ttyd Terminal-Binary aus der ersten Stufe rüber
COPY --from=ttyd /usr/bin/ttyd /usr/bin/ttyd

# Installiert die absolut notwendigen System-Bibliotheken für ttyd unter Alpine
RUN apk add --no-cache libuv tini

WORKDIR /app

# KORREKTUR: Universeller Suchpfad! Sucht rekursiv nach JEDER .jar im Projekt, 
# völlig egal ob sie in out/artifacts, artifacts oder direkt im Hauptordner liegt.
COPY **/B5_Project.jar /app/game.jar

# Port-Freigabe für Render
EXPOSE 8080

# Tini sorgt für ein stabiles Starten im Web mit iFrame-Freigabe (-I)
ENTRYPOINT ["/sbin/tini", "--"]
CMD ["ttyd", "-p", "8080", "-I", "java", "-jar", "/app/game.jar"]
